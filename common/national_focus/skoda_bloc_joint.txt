# A race for Skoda's order book. Any bloc country can take a focus; like the Commonwealth joint branch it then
# completes for the bloc members in the taker's faction, so each focus (and the other half of an either/or pick)

shared_focus = {
	id = SKODA_arsenal_of_the_danube
	icon = GFX_focus_generic_military_industry

	prerequisite = { focus = SKODA_seat_on_the_board }

	relative_position_id = SKODA_the_skoda_bloc
	x = 0
	y = 4

	cost = 5

	allow_branch = { SKODA_bloc_country = yes }

	# Only checked when starting: whoever started first keeps going.
	cancel_if_invalid = no
	continue_if_invalid = no
	available_if_capitulated = yes

	ai_will_do = { factor = 1 }
	# Locked for the whole bloc once any other bloc country is working on it or has finished it (no tooltip).
	available = {
		hidden_trigger = {
			NOT = {
				any_other_country = {
					SKODA_bloc_country = yes
					OR = {
						has_completed_focus = SKODA_arsenal_of_the_danube
						focus_progress = { focus = SKODA_arsenal_of_the_danube progress > 0 }
					}
				}
			}
		}
	}

	completion_reward = {
		custom_effect_tooltip = SKODA_chairman_tt
		custom_effect_tooltip = SKODA_arsenal_others_tt
		hidden_effect = {
		# completion_reward may run once per participant: only the first run hands out the ideas
		if = {
			limit = { NOT = { has_global_flag = SKODA_arsenal_done } }
			set_global_flag = SKODA_arsenal_done
			# Find the biggest shareholder (ties go to the earlier country in SLO, HUN, ROM, BUL).
			set_variable = { global.SKODA_shares_SLO = -1 }
			if = {
				limit = { country_exists = SLO }
				SLO = { set_variable = { global.SKODA_shares_SLO = SKODA_shares } }
			}
			set_variable = { global.SKODA_shares_HUN = -1 }
			if = {
				limit = { country_exists = HUN }
				HUN = { set_variable = { global.SKODA_shares_HUN = SKODA_shares } }
			}
			set_variable = { global.SKODA_shares_ROM = -1 }
			if = {
				limit = { country_exists = ROM }
				ROM = { set_variable = { global.SKODA_shares_ROM = SKODA_shares } }
			}
			set_variable = { global.SKODA_shares_BUL = -1 }
			if = {
				limit = { country_exists = BUL }
				BUL = { set_variable = { global.SKODA_shares_BUL = SKODA_shares } }
			}
			if = {
				limit = {
					check_variable = { global.SKODA_shares_SLO > -1 }
					NOT = { check_variable = { global.SKODA_shares_HUN > global.SKODA_shares_SLO } }
					NOT = { check_variable = { global.SKODA_shares_ROM > global.SKODA_shares_SLO } }
					NOT = { check_variable = { global.SKODA_shares_BUL > global.SKODA_shares_SLO } }
				}
				SLO = {
					add_ideas = SKODA_chairman_of_the_board
					set_country_flag = SKODA_chairman
				}
				set_global_flag = SKODA_chairman_SLO
			}
			else_if = {
				limit = {
					check_variable = { global.SKODA_shares_HUN > -1 }
					NOT = { check_variable = { global.SKODA_shares_ROM > global.SKODA_shares_HUN } }
					NOT = { check_variable = { global.SKODA_shares_BUL > global.SKODA_shares_HUN } }
					check_variable = { global.SKODA_shares_HUN > global.SKODA_shares_SLO }
				}
				HUN = {
					add_ideas = SKODA_chairman_of_the_board
					set_country_flag = SKODA_chairman
				}
				set_global_flag = SKODA_chairman_HUN
			}
			else_if = {
				limit = {
					check_variable = { global.SKODA_shares_ROM > -1 }
					NOT = { check_variable = { global.SKODA_shares_BUL > global.SKODA_shares_ROM } }
					check_variable = { global.SKODA_shares_ROM > global.SKODA_shares_SLO }
					check_variable = { global.SKODA_shares_ROM > global.SKODA_shares_HUN }
				}
				ROM = {
					add_ideas = SKODA_chairman_of_the_board
					set_country_flag = SKODA_chairman
				}
				set_global_flag = SKODA_chairman_ROM
			}
			else_if = {
				limit = {
					check_variable = { global.SKODA_shares_BUL > -1 }
					check_variable = { global.SKODA_shares_BUL > global.SKODA_shares_SLO }
					check_variable = { global.SKODA_shares_BUL > global.SKODA_shares_HUN }
					check_variable = { global.SKODA_shares_BUL > global.SKODA_shares_ROM }
				}
				BUL = {
					add_ideas = SKODA_chairman_of_the_board
					set_country_flag = SKODA_chairman
				}
				set_global_flag = SKODA_chairman_BUL
			}
			# every other bloc country holding at least one share gets Arsenal, whatever faction it is in
			every_country = {
				limit = {
					SKODA_bloc_country = yes
					check_variable = { SKODA_shares > 0 }
					NOT = { has_country_flag = SKODA_chairman }
				}
				add_ideas = SKODA_arsenal_of_the_danube_idea
			}
		}
		}
	}
}
