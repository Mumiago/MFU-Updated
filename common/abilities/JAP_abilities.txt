ability = {
	jap_last_stand = {
		name = ABILITY_LAST_STAND
		desc = ABILITY_LAST_STAND_DESC
		
		sound_effect = command_power_ability_offensive
		
		type = army_leader
		
		allowed = {
			OR = {
				hidden_trigger = { has_character_flag = { flag = MFU_on_jap_last_stand days < 2 } }
				AND = {
					is_leading_army_group = no
					is_border_war = no
					OWNER = {
						OR = {
							has_completed_focus = JAP_the_old_guard
						}
					}
					custom_trigger_tooltip = {
						tooltip = MFU_ability_no_force_attack_on_tt
						NOT = { MFU_force_attack_type_on = yes }
					}
					custom_trigger_tooltip = {
						tooltip = MFU_ability_no_org_shield_on_tt
						NOT = { MFU_org_shield_ability_on = yes }
					}
				}
			}
		}

		cancelable = yes
		cost = 0
		daily_cost = 0.017
		duration = -1
		one_time_effect = {
			hidden_effect = { set_character_flag = MFU_on_jap_last_stand }
		}
		daily_effect = {
			custom_effect_tooltip = MFU_ability_daily_cost_tt
			hidden_effect = {
				clr_character_flag = MFU_on_jap_last_stand
				set_character_flag = MFU_on_jap_last_stand
			}
		}
		end_effect = {
			hidden_effect = { clr_character_flag = MFU_on_jap_last_stand }
		}
		unit_modifiers = {
			defence = 0.2
			combat_entrenchment = 0.2
			org_damage_multiplier = -0.65
			str_damage_multiplier = 1
			war_support_reduction_on_damage = 0.2
		}

		ai_will_do = {
			factor = -1
			modifier = {
				FROM = {
					has_war_support > 0.4
				}
				
				check_variable = { num_units_defensive_combats > 6 }
				
				set_temp_variable = { temp = avg_defensive_combat_status }
				check_variable = { temp < 0.40 }
				check_variable = { ai_random > temp }
				
				add = 2
			}
		}
	}
}
