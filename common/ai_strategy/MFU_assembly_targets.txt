# MFU: teach AI to build tank/air assemblies so it has tank_production/air_production capacity.
# Scaled by military-factory count; one tier active per country. Targets are the tuning knob.
# Anchor: ~5 tank / ~5 air assemblies at ~150 mil factories (~1 assembly per ~30 mil factories).

MFU_assembly_tier_1 = {
	allowed = { always = yes }
	enable = {
		is_ai = yes
		num_of_military_factories < 20
	}
	ai_strategy = { type = building_target id = tank_assembly value = 1 }
	ai_strategy = { type = building_target id = air_assembly value = 1 }
}

# Construction priority: raise the AI's build weight for assemblies (internal default is 1)
# so it actually constructs them toward the tiered building_target above, instead of only
# ever building factories. Self-limited by the building_target count. Weight is the tuning knob:
# lower it if the AI over-invests in assemblies at the expense of factories.
MFU_assembly_build_priority = {
	allowed = { always = yes }
	enable = { is_ai = yes }
	ai_strategy = { type = build_building id = tank_assembly value = 10 }
	ai_strategy = { type = build_building id = air_assembly value = 10 }
}
