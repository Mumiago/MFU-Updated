# Afrikakorps scripted triggers (GER scope; used by the GER_africa_korps decision and the generated spawn effect,
# so ROOT is Germany). A city is usable when its state is held by Germany, a German ally (an allied Vichy included) or Italy.
# A Vichy-held city does not count by itself: that is neutral French territory. Tunis is also skipped while Tunisia is
# still a demilitarized zone (French occupation).
MFU_ak_city_tunis = {
	458 = {
		is_demilitarized_zone = no
		OR = {
			is_controlled_by = GER
			is_controlled_by = ITA
			is_controlled_by_ROOT_or_ally = yes
		}
	}
}
