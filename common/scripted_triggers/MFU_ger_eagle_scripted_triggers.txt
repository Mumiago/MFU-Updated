# Country scope. Japan's side: Japan, its subjects, its faction-mates that are not Germany's, and its known free allies.
GER_eagle_japan_side = {
	OR = {
		tag = JAP
		is_subject_of = JAP
		AND = {
			is_in_faction_with = JAP
			NOT = { tag = GER }
			NOT = { is_in_faction_with = GER }
		}
		AND = {
			is_in_faction_with = JAP
			OR = {
				original_tag = MAN
				original_tag = RCC
				original_tag = MEN
				original_tag = SIA
				original_tag = KOR
				original_tag = PHI
			}
		}
	}
}

# Country scope. Germany, its faction and its subjects, without Japan's side.
GER_eagle_german_side = {
	OR = {
		tag = GER
		is_in_faction_with = GER
		is_subject_of = GER
	}
	NOT = { GER_eagle_japan_side = yes }
}

# Country scope. Everyone at war with Germany, plus the Soviet Union at any time.
GER_eagle_allied_side = {
	OR = {
		tag = SOV
		has_war_with = GER
	}
	NOT = { GER_eagle_german_side = yes }
}
