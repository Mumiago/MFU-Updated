# MFU: permanently disable the "Ask for air base access" and "Offer air base access"
# diplomatic actions. HOI4 has no scripted way to hide a hardcoded diplomatic action
# from the list, so these engine-recognized enable triggers force it to always be
# unavailable (greyed, red X) with an explanatory tooltip.

DIPLOMACY_AIR_BASE_ACCESS_ENABLE_TRIGGER = {
	custom_trigger_tooltip = {
		tooltip = MFU_AIR_BASE_ACCESS_DISABLED_TT
		always = no
	}
}

DIPLOMACY_OFFER_AIR_BASE_ACCESS_ENABLE_TRIGGER = {
	custom_trigger_tooltip = {
		tooltip = MFU_AIR_BASE_ACCESS_DISABLED_TT
		always = no
	}
}
