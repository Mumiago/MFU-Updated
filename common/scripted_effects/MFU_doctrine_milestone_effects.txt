# Always evaluate the explicitly excluded major group first.
MFU_award_tier1_combined_arms_idea = {
    if = {
        limit = { MFU_is_combined_arms_major = yes }
        remove_ideas = combined_arms_bonus_minor_tier_1
        add_ideas = combined_arms_bonus1
    }
    else = {
        remove_ideas = combined_arms_bonus1
        add_ideas = combined_arms_bonus_minor_tier_1
    }

    # One Land Warfare breakthrough for the listed minors after both Tier I combat tracks, once per country.
    if = {
        limit = {
            OR = {
                tag = AST
                tag = BUL
                tag = CAN
                tag = CHI
                tag = EFR
                tag = FRA
                tag = HOL
                tag = HUN
                tag = MAN
                tag = NZL
                tag = POL
                tag = RAJ
                tag = ROM
                tag = SAF
                tag = SLO
                tag = SPR
            }
            MFU_is_combined_arms_major = no

            has_country_flag = finished_tier1_inf_mastery
            has_country_flag = finished_tier1_tank_mastery
            NOT = { has_country_flag = MFU_tier1_land_breakthrough_awarded }
        }
        set_country_flag = MFU_tier1_land_breakthrough_awarded
        add_breakthrough_points = {
            specialization = specialization_land
            value = 1
        }
    }
}
