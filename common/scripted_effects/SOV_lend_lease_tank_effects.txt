# Cost follows designer slot order.
# Flat module costs are added, then engine and armor-type multipliers hit the running total.
# Sloping, radio, and the special module are added after those multipliers.
# Engine and armor upgrade levels are flat IC added at the end: 0.1 and 0.25 per level.

SOV_ll_calculate_tank_cost = {
	set_variable = { SOV_ll_tank_cost = SOV_ll_chassis_cost }
	add_to_variable = { SOV_ll_tank_cost = SOV_ll_gun_cost }
	add_to_variable = { SOV_ll_tank_cost = SOV_ll_turret_cost }
	add_to_variable = { SOV_ll_tank_cost = SOV_ll_susp_cost }
	add_to_variable = { SOV_ll_tank_cost = SOV_ll_engine_cost }
	set_temp_variable = { SOV_ll_mult = 1 }
	add_to_temp_variable = { SOV_ll_mult = SOV_ll_engine_mult }
	multiply_variable = { var = SOV_ll_tank_cost value = SOV_ll_mult }
	set_temp_variable = { SOV_ll_mult = 1 }
	add_to_temp_variable = { SOV_ll_mult = SOV_ll_armor_mult }
	multiply_variable = { var = SOV_ll_tank_cost value = SOV_ll_mult }
	add_to_variable = { SOV_ll_tank_cost = SOV_ll_slope_cost }
	add_to_variable = { SOV_ll_tank_cost = SOV_ll_radio_cost }
	add_to_variable = { SOV_ll_tank_cost = SOV_ll_special_cost }
	set_temp_variable = { SOV_ll_upgrade_ic = SOV_ll_engine_level }
	multiply_temp_variable = { var = SOV_ll_upgrade_ic value = 0.1 }
	add_to_variable = { SOV_ll_tank_cost = SOV_ll_upgrade_ic }
	set_temp_variable = { SOV_ll_upgrade_ic = SOV_ll_armor_level }
	multiply_temp_variable = { var = SOV_ll_upgrade_ic value = 0.25 }
	add_to_variable = { SOV_ll_tank_cost = SOV_ll_upgrade_ic }
}
