# Romania: the Camarilla corruption meter (King path)
# ROM_camarilla runs 0-100. Higher corruption = more political power, less stability.

ROM_camarilla_update = {
	if = {
		limit = { has_country_flag = ROM_camarilla_capped }
		clamp_variable = { var = ROM_camarilla min = 0 max = 60 }
	}
	else = {
		clamp_variable = { var = ROM_camarilla min = 0 max = 100 }
	}
	set_variable = { ROM_cam_pp = ROM_camarilla }
	multiply_variable = { ROM_cam_pp = 0.002 }
	set_variable = { ROM_cam_stab = ROM_camarilla }
	multiply_variable = { ROM_cam_stab = -0.003 }
	force_update_dynamic_modifier = yes
}

# Romanian Industry and the Reich: every decision raises consumer goods by 2% for 90 days, stacking
ROM_industry_bargaining_cost = {
	custom_effect_tooltip = ROM_ib_bargaining_tt
	hidden_effect = {
		add_to_variable = { ROM_ib_cg = 0.02 }
		if = {
			limit = { NOT = { has_dynamic_modifier = { modifier = ROM_industry_bargaining } } }
			add_dynamic_modifier = { modifier = ROM_industry_bargaining }
		}
		force_update_dynamic_modifier = yes
		country_event = { id = rom_rework.31 days = 90 }
	}
}
