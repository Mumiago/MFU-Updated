# Call inside hidden_effect. Safe from any German focus, any order, any number of times.
GER_air_command_ensure_active = {
	if = {
		limit = {
			NOT = { has_dynamic_modifier = { modifier = GER_air_command_dm } }
		}
		set_variable = { GER_aircmd_cas_org = 0 }
		set_variable = { GER_aircmd_ground_attack = 0 }
		add_dynamic_modifier = { modifier = GER_air_command_dm }
	}
	if = {
		limit = {
			NOT = { has_idea = GER_luftwaffe_command_idea }
		}
		add_ideas = GER_luftwaffe_command_idea
	}
}

GER_air_command_refresh = {
	force_update_dynamic_modifier = yes
}

GER_4yp_stockpile_light_fighters = {
	if = {
		limit = { OR = { has_tech = ger_jet_fighter_1945 has_tech = jet_fighter_1946 } }
		add_equipment_to_stockpile = { type = jet_fighter_equipment_1945 amount = $AMOUNT$ producer = GER }
	}
	else_if = {

		add_equipment_to_stockpile = { type = fighter_equipment_1944 amount = $AMOUNT$ producer = GER }
	}
	else_if = {
		limit = { has_tech = ger_fighter_1943 }
		add_equipment_to_stockpile = { type = fighter_equipment_1943 amount = $AMOUNT$ producer = GER }

	else_if = {

		add_equipment_to_stockpile = { type = fighter_equipment_1942 amount = $AMOUNT$ producer = GER }
	}
	else_if = {
		limit = { has_tech = ger_fighter_1941 }
		add_equipment_to_stockpile = { type = fighter_equipment_1941 amount = $AMOUNT$ producer = GER }
	}
	else_if = {
		limit = { OR = { has_tech = ger_fighter_1940 has_tech = fighter_1940 } }
		add_equipment_to_stockpile = { type = fighter_equipment_1940 amount = $AMOUNT$ producer = GER }
	}
	else_if = {
		limit = { OR = { has_tech = ger_fighter_1938 has_tech = fighter_1938 } }
		add_equipment_to_stockpile = { type = fighter_equipment_1938 amount = $AMOUNT$ producer = GER }

	else_if = {
		limit = { OR = { has_tech = ger_fighter_1936 has_tech = fighter_1936 } }
