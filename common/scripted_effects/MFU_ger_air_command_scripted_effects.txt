# Call inside hidden_effect. Safe from any German focus, any order, any number of times.
GER_air_command_ensure_active = {
	if = {
		limit = {
			NOT = { has_dynamic_modifier = { modifier = GER_air_command_dm } }
		}
		set_variable = { GER_aircmd_cas_org = 0 }
		set_variable = { GER_aircmd_ground_attack = 0 }
		set_variable = { GER_aircmd_xp = 0 }
		set_variable = { GER_aircmd_xp_factor = 0 }
		set_variable = { GER_aircmd_mission_xp = 0 }
		set_variable = { GER_aircmd_training_xp = 0 }
		set_variable = { GER_aircmd_ace_chance = 0 }
		set_variable = { GER_aircmd_mission_eff = 0 }
		set_variable = { GER_aircmd_power_projection = 0 }
		set_variable = { GER_aircmd_home_defence = 0 }
		add_dynamic_modifier = { modifier = GER_air_command_dm }
	}
	if = {
		limit = {
			NOT = {
				OR = {
					has_idea = GER_luftwaffe_command_idea
					has_idea = GER_luftwaffe_command_2_idea
				}
			}
		}
		add_ideas = GER_luftwaffe_command_idea
	}
}

GER_air_command_refresh = {
	force_update_dynamic_modifier = yes
}

# Air innovation 2: swaps the spirit for the version with the plane cost bonus. Safe to call twice.
GER_air_command_upgrade = {
	GER_air_command_ensure_active = yes
	if = {
		limit = { has_idea = GER_luftwaffe_command_idea }
		swap_ideas = {
			remove_idea = GER_luftwaffe_command_idea
			add_idea = GER_luftwaffe_command_2_idea
		}
	}
}

# Luftwaffe workers mission (focus GER_forced_labor_air). Demand starts at 30000, rises 2000 per cycle, no cap.
# Each cycle the Air Ministry takes ALL workers (GER_num_arbeitseinsatz). Step = round(10 * (demand - workers) / demand), clamped -20..+10.
# The step is the change of plane production cost in percent: + = dearer (cancels the cut of the spirit), - = cheaper.
GER_air_workers_start = {
	set_variable = { GER_air_quota = 30000 }
	set_variable = { GER_air_cost_step = 0 }
	set_variable = { GER_air_last_eaten = 0 }
	set_variable = { GER_air_last_quota = 0 }
	set_country_flag = GER_air_workers_on
	activate_mission = GER_air_workers_mission
}

GER_air_workers_apply_ladder = {
	if = {
		limit = { has_idea = GER_air_cost_up_1 }
		remove_ideas = GER_air_cost_up_1
	}
	if = {
		limit = { has_idea = GER_air_cost_up_2 }
		remove_ideas = GER_air_cost_up_2
	}
	if = {
		limit = { has_idea = GER_air_cost_up_4 }
		remove_ideas = GER_air_cost_up_4
	}
	if = {
		limit = { has_idea = GER_air_cost_up_8 }
		remove_ideas = GER_air_cost_up_8
	}
	if = {
		limit = { has_idea = GER_air_cost_up_16 }
		remove_ideas = GER_air_cost_up_16
	}
	if = {
		limit = { has_idea = GER_air_cost_down_1 }
		remove_ideas = GER_air_cost_down_1
	}
	if = {
		limit = { has_idea = GER_air_cost_down_2 }
		remove_ideas = GER_air_cost_down_2
	}
	if = {
		limit = { has_idea = GER_air_cost_down_4 }
		remove_ideas = GER_air_cost_down_4
	}
	if = {
		limit = { has_idea = GER_air_cost_down_8 }
		remove_ideas = GER_air_cost_down_8
	}
	if = {
		limit = { has_idea = GER_air_cost_down_16 }
		remove_ideas = GER_air_cost_down_16
	}
	set_temp_variable = { rest = GER_air_cost_step }
	if = {
		limit = { check_variable = { rest > 0 } }
		if = {
			limit = { check_variable = { rest > 15 } }
			add_ideas = GER_air_cost_up_16
			add_to_temp_variable = { rest = -16 }
		}
		if = {
			limit = { check_variable = { rest > 7 } }
			add_ideas = GER_air_cost_up_8
			add_to_temp_variable = { rest = -8 }
		}
		if = {
			limit = { check_variable = { rest > 3 } }
			add_ideas = GER_air_cost_up_4
			add_to_temp_variable = { rest = -4 }
		}
		if = {
			limit = { check_variable = { rest > 1 } }
			add_ideas = GER_air_cost_up_2
			add_to_temp_variable = { rest = -2 }
		}
		if = {
			limit = { check_variable = { rest > 0 } }
			add_ideas = GER_air_cost_up_1
			add_to_temp_variable = { rest = -1 }
		}
	}
	else_if = {
		limit = { check_variable = { rest < 0 } }
		multiply_temp_variable = { rest = -1 }
		if = {
			limit = { check_variable = { rest > 15 } }
			add_ideas = GER_air_cost_down_16
			add_to_temp_variable = { rest = -16 }
		}
		if = {
			limit = { check_variable = { rest > 7 } }
			add_ideas = GER_air_cost_down_8
			add_to_temp_variable = { rest = -8 }
		}
		if = {
			limit = { check_variable = { rest > 3 } }
			add_ideas = GER_air_cost_down_4
			add_to_temp_variable = { rest = -4 }
		}
		if = {
			limit = { check_variable = { rest > 1 } }
			add_ideas = GER_air_cost_down_2
			add_to_temp_variable = { rest = -2 }
		}
		if = {
			limit = { check_variable = { rest > 0 } }
			add_ideas = GER_air_cost_down_1
			add_to_temp_variable = { rest = -1 }
		}
	}
}

GER_air_workers_cycle_end = {
	set_variable = { GER_air_last_eaten = GER_num_arbeitseinsatz }
	set_variable = { GER_air_last_quota = GER_air_quota }
	set_temp_variable = { gap = GER_air_quota }
	subtract_from_temp_variable = { gap = GER_num_arbeitseinsatz }
	divide_temp_variable = { gap = GER_air_quota }
	multiply_temp_variable = { gap = 10 }
	round_temp_variable = gap
	clamp_temp_variable = { var = gap min = -20 max = 10 }
	set_variable = { GER_air_cost_step = gap }
	set_variable = { GER_num_arbeitseinsatz = 0 }
	GER_air_workers_apply_ladder = yes
	add_to_variable = { GER_air_quota = 2000 }
	country_event = { id = ger_air_workers.1 }
}
