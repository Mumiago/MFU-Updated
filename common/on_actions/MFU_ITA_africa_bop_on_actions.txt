on_actions = {
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					FROM.FROM = {
						ITA_africa_bop_state = yes
						NOT = { has_state_flag = ITA_africa_bop_lost }
					}
					ROOT = { ITA_africa_bop_allied_country = yes }
					FROM = { ITA_africa_bop_axis_country = yes }
					country_exists = ITA
					ITA = { has_power_balance = { id = ITA_power_balance } }
				}
				FROM.FROM = { set_state_flag = ITA_africa_bop_lost }
				ITA = { ITA_africa_bop_apply_loss = yes }
			}
			else_if = {
				limit = {
					FROM.FROM = { has_state_flag = ITA_africa_bop_lost }
					ROOT = { ITA_africa_bop_axis_country = yes }
				}
				FROM.FROM = { clr_state_flag = ITA_africa_bop_lost }
				ITA = { ITA_africa_bop_apply_refund = yes }
			}
		}
	}
}
