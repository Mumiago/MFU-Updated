on_actions = {

	on_weekly_JAP = {
		effect = {
			every_country = {
				limit = {
					OR = {
						tag = JAP
						tag = MAN
						tag = RCC
					}
				}
				every_controlled_state = {
					limit = {
						is_core_of = CHI
						compliance > 50
						NOT = { has_state_flag = CHI_assassinations_cleared_flag }
						NOT = { has_dynamic_modifier = { modifier = CHI_pacified_by_japan } }
					}
					add_dynamic_modifier = { modifier = CHI_pacified_by_japan }
				}
			}
		}
	}

	on_state_control_changed = { #ROOT is new controller #FROM is old controller #FROM.FROM is state ID
		effect = {
			if = {
				limit = {
					FROM.FROM = { has_dynamic_modifier = { modifier = CHI_pacified_by_japan } }
					NOT = {
						ROOT = {
							OR = {
								tag = JAP
								tag = MAN
								tag = RCC
							}
						}
					}
				}
				FROM.FROM = { remove_dynamic_modifier = { modifier = CHI_pacified_by_japan } }
			}
		}
	}
}
