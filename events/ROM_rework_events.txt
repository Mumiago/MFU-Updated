# The Camarilla: weekly corruption tick
country_event = {
	id = rom_rework.20
	hidden = yes
	is_triggered_only = yes

	trigger = {
		original_tag = ROM
		NOT = { has_country_flag = ROM_camarilla_over }
	}

	immediate = {
		if = {
			limit = { NOT = { has_country_flag = ROM_camarilla_capped } }
			add_to_variable = { ROM_camarilla = 1 }
		}
		if = {
			limit = { any_character = { is_character = ROM_elena_lupescu is_hired_as_advisor = yes } }
			add_to_variable = { ROM_camarilla = 1 }
		}
		if = {
			limit = { any_character = { is_character = ROM_ernest_urdareanu is_hired_as_advisor = yes } }
			add_to_variable = { ROM_camarilla = 1 }
		}
		ROM_camarilla_update = yes
		if = {
			limit = { check_variable = { ROM_camarilla > 99 } }
			country_event = { id = rom_rework.22 }
		}
		else_if = {
			limit = {
				check_variable = { ROM_camarilla > 74 }
				NOT = { has_country_flag = ROM_camarilla_scandal }
			}
			country_event = { id = rom_rework.21 }
		}
		if = {
			limit = { check_variable = { ROM_camarilla < 50 } }
			clr_country_flag = ROM_camarilla_scandal
		}
		if = {
			limit = {
				NOT = { has_country_flag = ROM_camarilla_event_cooldown }
				check_variable = { ROM_camarilla > 49 }
			}
			random = { chance = 12 ROM_camarilla_random_event = yes }
		}
		else_if = {
			limit = { NOT = { has_country_flag = ROM_camarilla_event_cooldown } }
			random = { chance = 7 ROM_camarilla_random_event = yes }
		}
		country_event = { id = rom_rework.20 days = 7 }
	}
}

# Farm paths: every 3 months the Grain Economy's consumer goods worsen,
# and Support for the Peasantry's consumer goods improve
country_event = {
	id = rom_rework.30
	hidden = yes
	is_triggered_only = yes

	trigger = {
		original_tag = ROM
	}

	immediate = {
		if = {
			limit = { has_dynamic_modifier = { modifier = ROM_grain_economy } }
			add_to_variable = { ROM_ge_cg = 0.015 }
			clamp_variable = { var = ROM_ge_cg max = 0.15 }
		}
		if = {
			limit = { has_dynamic_modifier = { modifier = ROM_peasant_policy } }
			add_to_variable = { ROM_pp_consumer_goods = -0.01 }
			clamp_variable = { var = ROM_pp_consumer_goods min = -0.08 }
		}
		force_update_dynamic_modifier = yes
		country_event = { id = rom_rework.30 days = 91 }
	}
}
